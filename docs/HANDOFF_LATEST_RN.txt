from pathlib import Path
import re
pages = sorted(Path("pages").glob("*.html"))
page_names = {p.name for p in pages}
nav_refs = set()
for p in [Path("index.html")] + pages:
    text = p.read_text(errors="ignore")
    m = re.search(r"<nav[^>]*>(.*?)</nav>", text, re.S)
    if not m:
        continue
    for href in re.findall(r'href="([^"]+)"', m.group(1)):
        clean = href.split("?")[0].split("#")[0].replace("../", "").replace("pages/", "")
        if clean.endswith(".html"):
            nav_refs.add(clean)
print("PAGES NOT IN SIDE MENU:")
for x in sorted(page_names - nav_refs):
    print("-", x)
PY
